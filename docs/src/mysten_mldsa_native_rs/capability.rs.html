4//! Runtime CPU probe for the native backends.
5//!
6//! The vendored C asks "may I use this CPU feature?" through
7//! `mld_sys_check_capability()`; `native_dispatch.h` routes that call to the
8//! function below. Detection and caching come from the `cpufeatures` crate,
9//! so after the first call each answer is one atomic load.
10//!
11//! The capability numbers are positions in upstream's `mld_sys_cap` enum;
12//! the single_level_*.c shims verify them at compile time.
13
14use core::ffi::c_int;
15
16#[cfg(target_arch = "x86_64")]
17cpufeatures::new!(cpuid_avx2, "avx2");
18
19/// Called from the vendored C. 1 = supported, 0 = fall back to portable code.
20#[cfg(target_arch = "x86_64")]
21#[no_mangle]
22pub extern "C" fn mysten_mldsa_sys_check_capability(cap: c_int) -> c_int {
23    // 0 = MLD_SYS_CAP_X86_64_AVX2. cpufeatures also checks OSXSAVE/XCR0, so
24    // "yes" means the OS saves the YMM registers too.
25    (cap == 0 && cpuid_avx2::init().get()) as c_int
26}
27
28#[cfg(all(
29    target_arch = "aarch64",
30    any(target_os = "linux", target_os = "android", target_vendor = "apple")
31))]
32cpufeatures::new!(cpuid_sha3, "sha3");
33
34/// Called from the vendored C. 1 = supported, 0 = fall back to portable code.
35#[cfg(target_arch = "aarch64")]
36#[no_mangle]
37pub extern "C" fn mysten_mldsa_sys_check_capability(cap: c_int) -> c_int {
38    match cap {
39        // 1 = MLD_SYS_CAP_AARCH64_NEON: baseline on aarch64, always yes.
40        // Returning 0 would quietly turn the native backend into portable C.
41        1 => 1,
42        // 2 = MLD_SYS_CAP_AARCH64_SHA3: HWCAP on Linux/Android, sysctl on
43        // Apple. Other OSes have no probe and claim nothing.
44        #[cfg(any(target_os = "linux", target_os = "android", target_vendor = "apple"))]
45        2 => cpuid_sha3::init().get() as c_int,
46        _ => 0,
47    }
48}

50/// Stub for targets with no native backend; the C never asks here.
51#[cfg(not(any(target_arch = "x86_64", target_arch = "aarch64")))]
52#[no_mangle]
53pub extern "C" fn mysten_mldsa_sys_check_capability(_cap: c_int) -> c_int {
54    0
